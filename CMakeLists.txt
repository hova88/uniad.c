cmake_minimum_required(VERSION 3.16)
project(uniad_c VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(UNIAD_CUDA "Build CUDA backend" OFF)
option(UNIAD_SANITIZERS "Enable ASan/UBSan" OFF)
add_library(uniad_runtime STATIC src/uniad.c src/operators.c)
target_include_directories(uniad_runtime PUBLIC include)
target_compile_options(uniad_runtime PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(UNIAD_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(uniad_runtime PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(uniad_runtime PUBLIC -fsanitize=address,undefined)
endif()
if(UNIAD_CUDA)
  enable_language(CUDA)
  target_sources(uniad_runtime PRIVATE src/cuda_backend.cu)
  target_compile_definitions(uniad_runtime PUBLIC UA_WITH_CUDA=1)
  set_target_properties(uniad_runtime PROPERTIES CUDA_STANDARD 14)
endif()
add_executable(uniad src/cli.c)
target_link_libraries(uniad PRIVATE uniad_runtime m)
enable_testing()
add_executable(test_runtime tests/test_runtime.c)
target_include_directories(test_runtime PRIVATE src)
target_link_libraries(test_runtime PRIVATE uniad_runtime m)
add_test(NAME runtime COMMAND test_runtime)
